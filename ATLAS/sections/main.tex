\begin{frame}{Deep Dive: VBF Signature \& Backgrounds}
    \begin{columns}[T]
        
        % Left Column: Physics concepts
        \begin{column}{0.55\textwidth}
            \small
            \textbf{Background Discrimination:}
            \begin{itemize}
                \item \textbf{Irreducible (ggF + $\ge 2$ jets):} 
                Yields the exact same $H+jj$ final state. Cannot be rejected by particle identification. Separable \textit{only} via VBF-specific kinematics.
                \item \textbf{Reducible (VH):} 
                Different true final states. Suppressed via explicit vetos (e.g., extra leptons).
            \end{itemize}
            
            
            \textbf{Key VBF Signatures:}
            \begin{itemize}
                \item \textbf{$\Delta\eta_{jj}$:} Large pseudorapidity gap between the two forward-backward tagging jets.
                \item \textbf{$m_{jj}$:} Hard invariant mass spectrum.
            \end{itemize}
